<script>
    // Danh sách dữ liệu các loài hoa
    const flowerData = [
        {
            name: "Hoa Hướng Dương có màu đặc trưng nào?",
            correctColor: "Màu Vàng",
            options: ["Màu Vàng", "Màu Đỏ", "Màu Tím", "Màu Xanh"],
            image: "https://images.unsplash.com/photo-1597848212624-a19eb35e2651?auto=format&fit=crop&w=600&q=80"
        },
        {
            name: "Hoa Lavender (Oải Hương) có màu gì?",
            correctColor: "Màu Tím",
            options: ["Màu Hồng", "Màu Tím", "Màu Trắng", "Màu Vàng"],
            image: "https://images.unsplash.com/photo-1528183429752-a97d0bf99b5a?auto=format&fit=crop&w=600&q=80"
        },
        {
            name: "Hoa Cúc Tần Dưỡng trong tự nhiên thường có màu nổi bật nào?",
            correctColor: "Màu Trắng",
            options: ["Màu Đỏ", "Màu Cam", "Màu Trắng", "Màu Đen"],
            image: "https://images.unsplash.com/photo-1560717845-968823efbee1?auto=format&fit=crop&w=600&q=80"
        },
        {
            name: "Hoa Bỉ Ngạn truyền thuyết thường gắn liền với màu gì?",
            correctColor: "Màu Đỏ",
            options: ["Màu Xanh Băng", "Màu Đỏ", "Màu Vàng", "Màu Tím"],
            image: "https://images.unsplash.com/photo-1601004890684-d8cbf643f5f2?auto=format&fit=crop&w=600&q=80"
        },
        {
            name: "Hoa Cẩm Tú Cầu đổi sang màu gì khi trồng trong đất chua?",
            correctColor: "Màu Xanh Lam",
            options: ["Màu Vàng", "Màu Cam", "Màu Xanh Lam", "Màu Đỏ Thẫm"],
            image: "https://images.unsplash.com/photo-1508610048659-a06b669e3321?auto=format&fit=crop&w=600&q=80"
        }
    ];

    let currentQuestionIndex = 0;
    let score = 0;
    let highScore = localStorage.getItem("flowerGameHighScore") || 0;

    const imgElement = document.getElementById("flower-img");
    const nameElement = document.getElementById("flower-name");
    const optionsContainer = document.getElementById("options-container");
    const feedbackElement = document.getElementById("feedback-text");
    const nextBtn = document.getElementById("next-btn");
    const scoreElement = document.getElementById("current-score");
    const highScoreElement = document.getElementById("high-score");

    highScoreElement.innerText = highScore;

    function shuffleArray(array) {
        return array.sort(() => Math.random() - 0.5);
    }

    function loadQuestion() {
        feedbackElement.innerText = "";
        feedbackElement.className = "feedback";
        nextBtn.style.display = "none";

        const currentFlower = flowerData[currentQuestionIndex];
        imgElement.src = currentFlower.image;
        nameElement.innerText = currentFlower.name;

        optionsContainer.innerHTML = "";
        const shuffledOptions = shuffleArray([...currentFlower.options]);

        shuffledOptions.forEach(option => {
            const button = document.createElement("button");
            button.className = "btn-option";
            button.innerText = option;
            button.onclick = () => checkAnswer(option, currentFlower.correctColor);
            optionsContainer.appendChild(button);
        });
    }

    function checkAnswer(selectedOption, correctOption) {
        const buttons = optionsContainer.getElementsByClassName("btn-option");
        for (let btn of buttons) {
            btn.disabled = true; // Khóa các nút sau khi chọn
            if (btn.innerText === correctOption) {
                btn.style.backgroundColor = "#c8e6c9";
                btn.style.borderColor = "#2e7d32";
            } else if (btn.innerText === selectedOption) {
                btn.style.backgroundColor = "#ffcdd2";
                btn.style.borderColor = "#c62828";
            }
        }

        if (selectedOption === correctOption) {
            score += 10;
            scoreElement.innerText = score;
            feedbackElement.innerText = "Chính xác! 🎉";
            feedbackElement.className = "feedback correct";
            
            if (score > highScore) {
                highScore = score;
                localStorage.setItem("flowerGameHighScore", highScore);
                highScoreElement.innerText = highScore;
            }
        } else {
            feedbackElement.innerText = `Sai rồi! Đáp án là ${correctOption}.`;
            feedbackElement.className = "feedback wrong";
        }

        nextBtn.style.display = "block";
    }

    function loadNextQuestion() {
        currentQuestionIndex++;
        if (currentQuestionIndex < flowerData.length) {
            loadQuestion();
        } else {
            // Kết thúc lượt chơi, xáo trộn lại câu hỏi
            alert(`Bạn đã hoàn thành lượt chơi với ${score} điểm!`);
            currentQuestionIndex = 0;
            score = 0;
            scoreElement.innerText = score;
            shuffleArray(flowerData);
            loadQuestion();
        }
    }

    // Bắt đầu game
    shuffleArray(flowerData);
    loadQuestion();
</script>
